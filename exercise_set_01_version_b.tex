\subsection*{1.1 Equations}
Solve the following equations. State every admissible solution.

\begin{enumerate}[label=(\arabic*)]
  \item \(420-7x=0\).
  \item \(\dfrac{144}{x}=9\).
  \item \((x-7)^2=0\).
  \item \((x-3)^2=49\).
  \item \(8x-72+25x=324\).
\end{enumerate}

\subsection*{1.2 Derivatives}
\begin{enumerate}[label=(\arabic*),start=6]
  \item Define a derivative in one sentence. Give one microeconomic
  interpretation of a derivative.
  \item Find \(\dfrac{df}{dx}\) when \(f(x)=x^3\).
  \item Find \(\dfrac{df}{dx}\) when
  \(f(x)=\dfrac{2}{5}x^4+5x^2-8x\).
  \item Find \(\dfrac{df}{dx}\) when
  \(f(x)=\dfrac{2}{5}x^4+5x^2-8x+430\).
  \item Find \(\dfrac{df}{dx}\) when
  \(f(x)=3x^{18}-4x^{12}+7x^6-9x+100\).
  \item Find \(\dfrac{\partial f}{\partial x}\) when
  \(f(x,y)=3x^2+4xy+y^2-7\).
  \item Find \(\dfrac{\partial f}{\partial x}\) when
  \(f(x,y)=x^3y-2xy+5x\).
  \item Find \(\dfrac{\partial f}{\partial x}\) when
  \(f(x,y)=x^{2/5}y^{3/5}\).
\end{enumerate}

\section{Utility, budget constraints, and consumer choice}

\begin{enumerate}[label=\textbf{3.\alph*}]
  \item Elena has \euro72 to divide between museum tickets \(x\) and caf\'e
  lunches \(y\). Tickets cost \(p_x=\euro6\) and lunches cost
  \(p_y=\euro3\). Her utility function is
  \[
    u(x,y)=xy.
  \]
  \begin{enumerate}[label=(\roman*)]
    \item Write and draw Elena's budget constraint.
    \item Draw the indifference curves for \(u=36\), \(u=54\), and \(u=72\).
    \item Rewrite the budget line as \(y\) as a function of \(x\). Substitute
    it into utility to obtain \(u(x)\).
    \item Find the bundle that maximises utility using the first derivative.
    \item Recover the optimal quantity of \(y\). State the utility level at
    the optimum and verify the tangency condition.
  \end{enumerate}
\end{enumerate}

\begin{enumerate}[label=\textbf{4.\alph*}]
  \item A student chooses quantities of two goods \(x\) and \(y\). Income is
  \(I=\)\euro66, prices are \(p_x=\)\euro6 and \(p_y=\)\euro7, and utility is
  \[
    u(x,y)=xy+3y=y(x+3).
  \]
  \begin{enumerate}[label=(\roman*)]
    \item Write the budget line and draw the indifference curves for
    \(u=21\) and \(u=42\).
    \item Solve the consumer problem using \(MRS_{x,y}=p_x/p_y\) and the
    budget line.
    \item Verify the answer by substituting the budget line into utility and
    maximising the resulting one-variable function.
  \end{enumerate}
\end{enumerate}

\section{Rational preferences}

\begin{enumerate}[label=\textbf{5.\alph*}]
  \item State precisely the following axioms of rational preferences:
  completeness, transitivity, reflexivity, monotonicity (non-satiation),
  continuity, and convexity. For each axiom, give a short everyday consumption
  example that illustrates its economic meaning.
  \item A student ranks tacos as at least as good as pizza, pizza as at least
  as good as salad, and salad as strictly better than tacos. Which axiom is
  violated? Explain briefly.
\end{enumerate}

\section{Budget sets, relative prices, and opportunity cost}

\begin{enumerate}[label=\textbf{6.\alph*}]
  \item A student has income \(I=\)\euro30. Campus lunches are good \(x\) and
  cost \(p_x=\)\euro6; drinks are good \(y\) and cost \(p_y=\)\euro3.
  \begin{enumerate}[label=(\roman*)]
    \item Write the budget set and the budget line. Find both intercepts.
    \item What is the opportunity cost of one extra lunch in units of drinks?
    \item Suppose income rises to \(I=\euro42\), with prices unchanged.
    Describe or draw the change in the feasible set.
    \item Keeping income at \euro42, suppose the price of a drink rises to
    \euro6. Write the new budget line and explain the pivot relative to part
    (iii).
  \end{enumerate}
\end{enumerate}

\section{MRS, convexity, and demand classifications}

\begin{enumerate}[label=\textbf{9.\alph*}]
  \item Return to the utility function in Exercise 4,
  \(u(x,y)=y(x+3)\).
  \begin{enumerate}[label=(\roman*)]
    \item Derive \(MRS_{x,y}\).
    \item Along a positive indifference curve \(u(x,y)=\bar u>0\), show that
    \[
      MRS_{x,y}=\frac{\bar u}{(x+3)^2}.
    \]
    Does the MRS diminish as \(x\) increases? Explain how your result relates
    to convex indifference curves and the convexity axiom.
  \end{enumerate}
  \item Define normal (sometimes called superior) and inferior goods in terms of
  \(\partial q/\partial I\), holding prices fixed. For
  \(0\leq I\leq 400\), classify
  \[
    q_{\mathrm{bus}}(I)=12-0.02I
    \qquad\text{and}\qquad
    q_{\mathrm{coffee}}(I)=1+0.015I.
  \]
  \item Explain why a Giffen good must be inferior. Your answer should refer
  to the income and substitution effects of a price fall.
\end{enumerate}

\section{Aggregating individual demands}

\begin{enumerate}[label=\textbf{10.\alph*}]
  \item At a campus pop-up, two groups have weekly demand for reusable bottles
  given by
  \[
    q_A(p)=\max\{0,8-p\}
    \qquad\text{and}\qquad
    q_B(p)=\max\left\{0,5-\frac{1}{2}p\right\},
    \qquad p\geq0.
  \]
  \begin{enumerate}[label=(\roman*)]
    \item Construct the market demand \(Q(p)=q_A(p)+q_B(p)\) as a piecewise
    function.
    \item Identify each individual choke price, the interior kink in market
    demand, and the aggregate choke price.
    \item Sketch the market demand curve. Why do we add quantities at a common
    price rather than adding prices?
  \end{enumerate}
\end{enumerate}
